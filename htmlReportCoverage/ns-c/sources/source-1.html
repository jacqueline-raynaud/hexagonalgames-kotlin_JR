 @OptIn(ExperimentalMaterial3Api::class)
 @Composable
 fun AccountManagementScreen(
     modifier: Modifier = Modifier,
     onBackClick: () -> Unit
 ) {
     val viewModel : AccountManagementViewModel = hiltViewModel()
 
     Scaffold(
         modifier = modifier,
         topBar = {
             TopAppBar(
                 title = { Text(stringResource(id = R.string.account_management)) },
                 navigationIcon = {
                     IconButton(onClick = { onBackClick() })
                     {
                         Icon(
                             imageVector = Icons.AutoMirrored.Filled.ArrowBack,
                             contentDescription = stringResource(id = R.string.contentDescription_go_back)
                         )
                     }
                 }
             )
         }
     ) { contentPadding ->
         AccountManagement(
             modifier = Modifier.padding(contentPadding),
             onSignOutClicked = {
                 viewModel.signOut(onSuccess = onBackClick)
             },
             onDeleteAccountClicked = {
                 viewModel.deleteAccount(
                     onSuccess = { viewModel.signOut(onBackClick) },
                     onFailure = { exception ->
                         // TODO toast ou snackbar d'erreur voir si etat
                     }
                 )
             }
         )
     }
 }
